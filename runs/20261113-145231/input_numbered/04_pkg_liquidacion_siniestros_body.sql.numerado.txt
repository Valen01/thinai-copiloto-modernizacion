0001: CREATE OR REPLACE PACKAGE BODY pkg_liquidacion_siniestros AS
0002: /*******************************************************************************
0003:  * SISTEMA      : SIGSE - Sistema Integral de Gestion de Seguros
0004:  * MODULO       : SINIESTROS
0005:  * OBJETO       : PKG_LIQUIDACION_SINIESTROS (cuerpo)
0006:  * DESCRIPCION  : Implementacion del motor de liquidacion de siniestros.
0007:  *
0008:  * FLUJO GENERAL DE LIQUIDACION (pr_liquidar_siniestro)
0009:  * -----------------------------------------------------------------------------
0010:  *   1. Validaciones de procedencia (cualquier fallo => RECHAZADO):
0011:  *        a) Estado de la poliza.
0012:  *        b) Ocurrencia dentro de la vigencia.
0013:  *        c) Plazo de denuncia.
0014:  *        d) Cobertura contratada.
0015:  *        e) Periodo de carencia.
0016:  *        f) Morosidad de prima.
0017:  *   2. Calculo de la indemnizacion:
0018:  *        a) Monto base (tope por limite de cobertura).
0019:  *        b) Deducible (fijo / porcentual, minimo, beneficio VIP).
0020:  *        c) Monto cubierto (% de cobertura).
0021:  *        d) Penalizacion por siniestralidad.
0022:  *        e) Tope por suma asegurada remanente.
0023:  *        f) Retencion impositiva y monto neto.
0024:  *   3. Determinacion del estado final:
0025:  *        a) Alerta de fraude          => PENDIENTE_APROBACION.
0026:  *        b) Supera tope automatico    => PENDIENTE_APROBACION.
0027:  *        c) Caso contrario            => APROBADA.
0028:  *
0029:  * CODIGOS DE MOTIVO (LS_LIQUIDACIONES.MOTIVO)
0030:  * -----------------------------------------------------------------------------
0031:  *   LIQUIDACION_AUTOMATICA  Liquidacion aprobada sin intervencion.
0032:  *   SUPERA_TOPE_AUTOMATICO  Neto mayor al tope de aprobacion automatica.
0033:  *   SOSPECHA_FRAUDE         Siniestro con alerta de fraude.
0034:  *   POLIZA_NO_VIGENTE       Poliza cancelada o suspendida.
0035:  *   FUERA_DE_VIGENCIA       Ocurrencia fuera del periodo de vigencia.
0036:  *   DENUNCIA_EXTEMPORANEA   Denuncia presentada fuera de plazo.
0037:  *   COBERTURA_NO_CONTRATADA La poliza no tiene la cobertura reclamada.
0038:  *   EN_PERIODO_CARENCIA     Ocurrencia dentro del periodo de carencia.
0039:  *   PRIMA_EN_MORA           Poliza con prima pendiente de pago.
0040:  *   BAJO_DEDUCIBLE          El monto cubierto resultante es <= 0.
0041:  *   SUMA_ASEGURADA_AGOTADA  No queda suma asegurada disponible.
0042:  *
0043:  * MARCAS DE OBSERVACION (LS_LIQUIDACIONES.OBSERVACIONES, separadas por ';')
0044:  * -----------------------------------------------------------------------------
0045:  *   TOLERANCIA_MORA_VIP, LIMITE_COBERTURA, DEDUCIBLE_MINIMO, DESCUENTO_VIP,
0046:  *   PENALIZACION_SINIESTRALIDAD, TOPE_SUMA_ASEGURADA, RETENCION_APLICADA
0047:  *
0048:  * CONVENCION DE REDONDEO
0049:  * -----------------------------------------------------------------------------
0050:  *   Todos los importes calculados se redondean a 2 decimales con ROUND
0051:  *   (mitad hacia arriba) inmediatamente despues de cada operacion de
0052:  *   porcentaje.
0053:  ******************************************************************************/

0142:     -- FN_CALCULAR_DEDUCIBLE_V1
0143:     --   Calculo de deducible (version 2009).
0144:     --   Deducible fijo o porcentual sobre el monto reclamado, sin minimo.
0145:     ----------------------------------------------------------------------------
0146:     FUNCTION fn_calcular_deducible_v1 (
0147:         p_deducible_tipo   IN VARCHAR2,
0148:         p_deducible_valor  IN NUMBER,
0149:         p_monto_reclamado  IN NUMBER
0150:     ) RETURN NUMBER IS
0151:         v_deducible  NUMBER := 0;
0152:     BEGIN
0153:         IF p_deducible_tipo = 'FIJO' THEN

0155:         ELSIF p_deducible_tipo = 'PORCENTAJE' THEN
0156:             v_deducible := p_monto_reclamado * p_deducible_valor / 100;
0157:         END IF;
0158: 
0159:         RETURN TRUNC(v_deducible, 2);
0160: 
0161:         -- Ajuste por moneda extranjera
0162:         v_deducible := v_deducible * 1.0;
0163:         RETURN v_deducible;

0166:     ----------------------------------------------------------------------------
0167:     -- FN_CALCULAR_DEDUCIBLE
0168:     --   Calcula el deducible aplicable a un siniestro.
0169:     --
0170:     --   * Deducible FIJO       : se toma el valor fijo de la poliza.
0171:     --   * Deducible PORCENTAJE : % sobre el MONTO BASE (no sobre el reclamado),
0172:     --                            redondeado a 2 decimales. Si el resultado es
0173:     --                            menor al parametro DEDUCIBLE_MINIMO se aplica
0174:     --                            el minimo (marca DEDUCIBLE_MINIMO).
0175:     --   * Cliente VIP          : si el deducible resultante es > 0 se reduce en
0176:     --                            PCT_DESCUENTO_DED_VIP % (marca DESCUENTO_VIP).
0177:     --                            El descuento se aplica DESPUES del minimo.
0178:     ----------------------------------------------------------------------------
0179:     FUNCTION fn_calcular_deducible (
0180:         p_poliza         IN            ls_polizas%ROWTYPE,
0181:         p_categoria_cli  IN            ls_clientes.categoria%TYPE,
0182:         p_monto_base     IN            NUMBER,
0183:         p_observaciones  IN OUT NOCOPY VARCHAR2
0184:     ) RETURN NUMBER IS
0185:         v_deducible      NUMBER(14,2);
0186:         v_ded_minimo     NUMBER;
0187:         v_pct_desc_vip   NUMBER;
0188:     BEGIN
0189:         IF p_poliza.deducible_tipo = 'FIJO' THEN
0190:             v_deducible := p_poliza.deducible_valor;
0191:         ELSE
0192:             v_ded_minimo := fn_param_num('DEDUCIBLE_MINIMO');
0193:             v_deducible  := ROUND(p_monto_base * p_poliza.deducible_valor / 100, 2);
0194: 
0195:             IF v_deducible < v_ded_minimo THEN
0196:                 v_deducible := v_ded_minimo;
0197:                 pr_agregar_obs(p_observaciones, 'DEDUCIBLE_MINIMO');
0198:             END IF;
0199:         END IF;
0200: 
0201:         IF p_categoria_cli = c_cat_vip AND v_deducible > 0 THEN
0202:             v_pct_desc_vip := fn_param_num('PCT_DESCUENTO_DED_VIP');
0203:             v_deducible    := ROUND(v_deducible * (100 - v_pct_desc_vip) / 100, 2);
0204:             pr_agregar_obs(p_observaciones, 'DESCUENTO_VIP');
0205:         END IF;
0206: 
0207:         RETURN v_deducible;
0208:     END fn_calcular_deducible;

0325:     -- PR_LIQUIDAR_SINIESTRO  (publico)
0326:     ----------------------------------------------------------------------------
0327:     PROCEDURE pr_liquidar_siniestro (
0328:         p_siniestro_id   IN ls_siniestros.siniestro_id%TYPE,
0329:         p_fecha_proceso  IN DATE
0330:     ) IS
0331:         v_sin                ls_siniestros%ROWTYPE;
0332:         v_pol                ls_polizas%ROWTYPE;
0333:         v_cli                ls_clientes%ROWTYPE;
0334:         v_cob                ls_coberturas%ROWTYPE;
0335: 
0336:         v_dias_denuncia      NUMBER;
0337:         v_dias_max_denuncia  NUMBER;
0338:         v_dias_desde_inicio  NUMBER;
0339:         v_tolerancia_mora    NUMBER;
0340:         v_cant_siniestros    NUMBER;
0341:         v_total_indemnizado  NUMBER;
0342:         v_suma_remanente     NUMBER;
0343:         v_umbral_fraude      NUMBER;
0344: 
0345:         v_monto_base         NUMBER(14,2) := 0;
0346:         v_deducible          NUMBER(14,2) := 0;
0347:         v_cubierto           NUMBER(14,2) := 0;
0348:         v_retencion          NUMBER(14,2) := 0;
0349:         v_neto               NUMBER(14,2) := 0;
0350:         v_estado_liq         ls_liquidaciones.estado%TYPE;
0351:         v_motivo             ls_liquidaciones.motivo%TYPE;
0352:         v_observaciones      VARCHAR2(400);
0353: 
0354:         v_factor_inflacion   NUMBER;
0355:         v_contador_aux       PLS_INTEGER := 0;
0356:         v_fecha_proceso_ant  DATE;
0357:     BEGIN
0358:         ------------------------------------------------------------------------
0359:         -- Carga de datos del siniestro, poliza y cliente
0360:         ------------------------------------------------------------------------
0361:         SELECT *
0362:           INTO v_sin
0363:           FROM ls_siniestros
0364:          WHERE siniestro_id = p_siniestro_id
0365:            FOR UPDATE;
0366: 
0367:         IF v_sin.estado <> c_sin_registrado THEN
0368:             RAISE_APPLICATION_ERROR(-20001,
0369:                 'El siniestro ' || p_siniestro_id || ' no esta en estado REGISTRADO (' || v_sin.estado || ')');
0370:         END IF;
0371: 
0372:         SELECT * INTO v_pol FROM ls_polizas  WHERE poliza_id  = v_sin.poliza_id;
0373:         SELECT * INTO v_cli FROM ls_clientes WHERE cliente_id = v_pol.cliente_id;
0374: 
0375:         v_factor_inflacion := fn_param_num('FACTOR_INFLACION');
0376: 
0377:         IF c_modo_debug THEN
0378:             DBMS_OUTPUT.PUT_LINE('DEBUG siniestro=' || v_sin.siniestro_id ||
0379:                                  ' poliza=' || v_pol.numero_poliza ||
0380:                                  ' cliente=' || v_cli.nombre);
0381:         END IF;
0382: 
0383:         ------------------------------------------------------------------------
0384:         -- 1.a  Estado de la poliza: solo se liquidan polizas VIGENTES.
0385:         ------------------------------------------------------------------------
0386:         IF v_pol.estado <> c_poliza_vigente THEN
0387:             pr_rechazar(v_sin, p_fecha_proceso, 'POLIZA_NO_VIGENTE');
0388:             RETURN;
0389:         END IF;
0390: 
0391:         ------------------------------------------------------------------------
0392:         -- 1.b  La fecha de ocurrencia debe estar dentro de la vigencia
0393:         --      [fecha_inicio, fecha_fin] (ambos extremos inclusive).
0394:         ------------------------------------------------------------------------
0395:         IF v_sin.fecha_ocurrencia < v_pol.fecha_inicio
0396:            OR v_sin.fecha_ocurrencia > v_pol.fecha_fin
0397:         THEN
0398:             pr_rechazar(v_sin, p_fecha_proceso, 'FUERA_DE_VIGENCIA');
0399:             RETURN;
0400:         END IF;
0401: 
0402:         ------------------------------------------------------------------------
0403:         -- 1.c  Plazo de denuncia: dias entre ocurrencia y denuncia no puede
0404:         --      superar DIAS_MAX_DENUNCIA (VIDA: DIAS_MAX_DENUNCIA_VIDA).
0405:         --      El dia limite es inclusive.
0406:         ------------------------------------------------------------------------
0407:         v_dias_denuncia := TRUNC(v_sin.fecha_denuncia) - TRUNC(v_sin.fecha_ocurrencia);
0408: 
0409:         IF v_pol.ramo = c_ramo_vida THEN
0410:             v_dias_max_denuncia := fn_param_num('DIAS_MAX_DENUNCIA_VIDA');
0411:         ELSE
0412:             v_dias_max_denuncia := fn_param_num('DIAS_MAX_DENUNCIA');
0413:         END IF;
0414: 
0415:         IF v_dias_denuncia > v_dias_max_denuncia THEN
0416:             pr_rechazar(v_sin, p_fecha_proceso, 'DENUNCIA_EXTEMPORANEA');
0417:             RETURN;
0418:         END IF;
0419: 
0420:         ------------------------------------------------------------------------
0421:         -- 1.d  La cobertura reclamada debe estar contratada en la poliza.
0422:         ------------------------------------------------------------------------
0423:         BEGIN
0424:             SELECT *
0425:               INTO v_cob
0426:               FROM ls_coberturas
0427:              WHERE poliza_id     = v_sin.poliza_id
0428:                AND cod_cobertura = v_sin.cod_cobertura;
0429:         EXCEPTION
0430:             WHEN NO_DATA_FOUND THEN
0431:                 pr_rechazar(v_sin, p_fecha_proceso, 'COBERTURA_NO_CONTRATADA');
0432:                 RETURN;
0433:         END;
0434: 
0435:         ------------------------------------------------------------------------
0436:         -- 1.e  Periodo de carencia: la cobertura se activa cuando han
0437:         --      transcurrido al menos DIAS_CARENCIA dias desde el inicio de la
0438:         --      poliza hasta la ocurrencia.
0439:         ------------------------------------------------------------------------
0440:         v_dias_desde_inicio := TRUNC(v_sin.fecha_ocurrencia) - TRUNC(v_pol.fecha_inicio);
0441: 
0442:         IF v_dias_desde_inicio < v_cob.dias_carencia THEN
0443:             pr_rechazar(v_sin, p_fecha_proceso, 'EN_PERIODO_CARENCIA');
0444:             RETURN;
0445:         END IF;
0446: 
0447:         ------------------------------------------------------------------------
0448:         -- 1.f  Morosidad: con prima pendiente > 0 se rechaza, salvo clientes
0449:         --      VIP cuya deuda sea <= TOLERANCIA_MORA_VIP.
0450:         ------------------------------------------------------------------------
0451:         IF v_pol.prima_pendiente > 0 THEN
0452:             v_tolerancia_mora := fn_param_num('TOLERANCIA_MORA_VIP');
0453: 
0454:             IF v_cli.categoria = c_cat_vip
0455:                AND v_pol.prima_pendiente <= v_tolerancia_mora
0456:             THEN
0457:                 pr_agregar_obs(v_observaciones, 'TOLERANCIA_MORA_VIP');
0458:             ELSE
0459:                 pr_rechazar(v_sin, p_fecha_proceso, 'PRIMA_EN_MORA', v_observaciones);
0460:                 RETURN;
0461:             END IF;
0462:         END IF;
0463: 
0464:         -- Polizas suspendidas: registrar advertencia en bitacora
0465:         IF v_pol.estado = 'SUSPENDIDA' THEN
0466:             pr_log('WARN', v_sin.siniestro_id, 'Poliza suspendida: ' || v_pol.numero_poliza);
0467:             v_contador_aux := v_contador_aux + 1;
0468:         END IF;
0469: 
0470:         /*
0471:         -- SIN-0001 (2009): calculo original, reemplazado en SIN-0310
0472:         v_monto_base := v_sin.monto_reclamado;
0473:         v_deducible  := fn_calcular_deducible_v1(v_pol.deducible_tipo,
0474:                                                  v_pol.deducible_valor,
0475:                                                  v_sin.monto_reclamado);
0476:         v_cubierto   := v_monto_base - v_deducible;
0477:         IF v_cubierto > v_pol.suma_asegurada THEN
0478:             v_cubierto := v_pol.suma_asegurada;
0479:         END IF;
0480:         v_neto := v_cubierto;
0481:         */
0482: 
0483:         ------------------------------------------------------------------------
0484:         -- 2.a  Monto base: el menor entre lo reclamado y el limite de la
0485:         --      cobertura (marca LIMITE_COBERTURA si se topa).
0486:         ------------------------------------------------------------------------
0487:         IF v_sin.monto_reclamado > v_cob.limite_cobertura THEN
0488:             v_monto_base := v_cob.limite_cobertura;
0489:             pr_agregar_obs(v_observaciones, 'LIMITE_COBERTURA');
0490:         ELSE
0491:             v_monto_base := v_sin.monto_reclamado;
0492:         END IF;
0493: 
0494:         ------------------------------------------------------------------------
0495:         -- 2.b  Deducible (ver fn_calcular_deducible).
0496:         ------------------------------------------------------------------------
0497:         v_deducible := fn_calcular_deducible(v_pol, v_cli.categoria, v_monto_base, v_observaciones);
0498: 
0499:         ------------------------------------------------------------------------
0500:         -- 2.c  Monto cubierto = (monto base - deducible) * % cobertura.
0501:         --      Si el resultado es <= 0, se rechaza por BAJO_DEDUCIBLE.
0502:         ------------------------------------------------------------------------
0503:         v_cubierto := ROUND((v_monto_base - v_deducible) * v_cob.porcentaje_cobertura / 100, 2);
0504: 
0505:         IF v_cubierto <= 0 THEN
0506:             pr_rechazar(v_sin, p_fecha_proceso, 'BAJO_DEDUCIBLE', v_observaciones);
0507:             RETURN;
0508:         END IF;
0509: 
0510:         IF v_cubierto < 0 THEN
0511:             v_cubierto := 0;
0512:             RAISE e_monto_invalido;
0513:         END IF;
0514: 
0515:         ------------------------------------------------------------------------
0516:         -- 2.d  Siniestralidad: si la poliza registra MAX_SINIESTROS_ANUAL o
0517:         --      mas siniestros LIQUIDADOS con ocurrencia en los 365 dias
0518:         --      previos a la ocurrencia actual, se reduce el monto cubierto
0519:         --      en PCT_PENALIZACION_SINIEST %.
0520:         ------------------------------------------------------------------------
0521:         SELECT COUNT(*)
0522:           INTO v_cant_siniestros
0523:           FROM ls_siniestros s
0524:          WHERE s.poliza_id         = v_sin.poliza_id
0525:            AND s.siniestro_id     <> v_sin.siniestro_id
0526:            AND s.estado            = c_sin_liquidado
0527:            AND s.fecha_ocurrencia >= v_sin.fecha_ocurrencia - 365
0528:            AND s.fecha_ocurrencia <  v_sin.fecha_ocurrencia;
0529: 
0530:         IF v_cant_siniestros >= fn_param_num('MAX_SINIESTROS_ANUAL') THEN
0531:             v_cubierto := ROUND(v_cubierto * (100 - fn_param_num('PCT_PENALIZACION_SINIEST')) / 100, 2);
0532:             pr_agregar_obs(v_observaciones, 'PENALIZACION_SINIESTRALIDAD');
0533:         END IF;
0534: 
0535:         ------------------------------------------------------------------------
0536:         -- 2.e  Suma asegurada remanente = suma asegurada - suma de montos
0537:         --      cubiertos de liquidaciones APROBADAS o PENDIENTES de la poliza.
0538:         --      Si no hay remanente se rechaza; si el cubierto lo excede se
0539:         --      topa al remanente (marca TOPE_SUMA_ASEGURADA).
0540:         ------------------------------------------------------------------------
0541:         SELECT NVL(SUM(l.monto_cubierto), 0)
0542:           INTO v_total_indemnizado
0543:           FROM ls_liquidaciones l
0544:           JOIN ls_siniestros    s ON s.siniestro_id = l.siniestro_id
0545:          WHERE s.poliza_id = v_sin.poliza_id
0546:            AND l.estado IN (c_liq_aprobada, c_liq_pend_aprob);
0547: 
0548:         v_suma_remanente := v_pol.suma_asegurada - v_total_indemnizado;
0549: 
0550:         IF v_suma_remanente <= 0 THEN
0551:             pr_rechazar(v_sin, p_fecha_proceso, 'SUMA_ASEGURADA_AGOTADA', v_observaciones);
0552:             RETURN;
0553:         END IF;
0554: 
0555:         IF v_cubierto > v_suma_remanente THEN
0556:             v_cubierto := v_suma_remanente;
0557:             pr_agregar_obs(v_observaciones, 'TOPE_SUMA_ASEGURADA');
0558:         END IF;
0559: 
0560:         ------------------------------------------------------------------------
0561:         -- 2.f  Retencion impositiva: si el ramo no es VIDA y el monto
0562:         --      cubierto supera UMBRAL_RETENCION, se retiene TASA_RETENCION %.
0563:         --      Neto = cubierto - retencion.
0564:         ------------------------------------------------------------------------
0565:         IF v_pol.ramo <> c_ramo_vida
0566:            AND v_cubierto > fn_param_num('UMBRAL_RETENCION')
0567:         THEN
0568:             v_retencion := ROUND(v_cubierto * fn_param_num('TASA_RETENCION') / 100, 2);
0569:             pr_agregar_obs(v_observaciones, 'RETENCION_APLICADA');
0570:         ELSE
0571:             v_retencion := 0;
0572:         END IF;
0573: 
0574:         v_neto := v_cubierto - v_retencion;
0575: 
0576:         ------------------------------------------------------------------------
0577:         -- 3.  Estado final
0578:         --   a) Fraude: ocurrencia dentro de los primeros DIAS_ALERTA_FRAUDE
0579:         --      dias de la poliza y reclamo >= PCT_ALERTA_FRAUDE % del limite
0580:         --      de la cobertura => PENDIENTE_APROBACION / SOSPECHA_FRAUDE.
0581:         --   b) Neto > TOPE_AUTO_APROBACION
0582:         --      => PENDIENTE_APROBACION / SUPERA_TOPE_AUTOMATICO.
0583:         --   c) Resto => APROBADA / LIQUIDACION_AUTOMATICA.
0584:         --   La regla de fraude tiene prioridad sobre la de tope.
0585:         ------------------------------------------------------------------------
0586:         v_umbral_fraude := ROUND(v_cob.limite_cobertura * fn_param_num('PCT_ALERTA_FRAUDE') / 100, 2);
0587: 
0588:         IF v_dias_desde_inicio <= fn_param_num('DIAS_ALERTA_FRAUDE')
0589:            AND v_sin.monto_reclamado >= v_umbral_fraude
0590:         THEN
0591:             v_estado_liq := c_liq_pend_aprob;
0592:             v_motivo     := 'SOSPECHA_FRAUDE';
0593:         ELSIF v_neto > fn_param_num('TOPE_AUTO_APROBACION') THEN
0594:             v_estado_liq := c_liq_pend_aprob;
0595:             v_motivo     := 'SUPERA_TOPE_AUTOMATICO';
0596:         ELSE
0597:             v_estado_liq := c_liq_aprobada;
0598:             v_motivo     := 'LIQUIDACION_AUTOMATICA';
0599:         END IF;
0600: 
0601:         pr_registrar_resultado(
0602:             p_siniestro     => v_sin,
0603:             p_fecha_proceso => p_fecha_proceso,
0604:             p_monto_base    => v_monto_base,
0605:             p_deducible     => v_deducible,
0606:             p_cubierto      => v_cubierto,
0607:             p_retencion     => v_retencion,
0608:             p_neto          => v_neto,
0609:             p_estado_liq    => v_estado_liq,
0610:             p_motivo        => v_motivo,
0611:             p_observaciones => v_observaciones
0612:         );
0613: 
0614:         IF c_modo_debug THEN
0615:             pr_enviar_notificacion(v_sin.siniestro_id, v_estado_liq);
0616:         END IF;
0617:     END pr_liquidar_siniestro;

0619:     ----------------------------------------------------------------------------
0620:     -- PR_PROCESAR_LOTE  (publico)
0621:     ----------------------------------------------------------------------------
0622:     PROCEDURE pr_procesar_lote (
0623:         p_fecha_proceso  IN DATE DEFAULT TRUNC(SYSDATE)
0624:     ) IS
0625:         CURSOR cur_pendientes IS
0626:             SELECT s.siniestro_id, s.monto_reclamado
0627:               FROM ls_siniestros s
0628:              WHERE s.estado = c_sin_registrado
0629:                AND s.fecha_denuncia <= p_fecha_proceso
0630:              ORDER BY s.siniestro_id;
0631: 
0632:         v_estado_final     ls_siniestros.estado%TYPE;
0633:         v_cant_procesados  PLS_INTEGER  := 0;
0634:         v_cant_liquidados  PLS_INTEGER  := 0;
0635:         v_cant_rechazados  PLS_INTEGER  := 0;
0636:         v_cant_pendientes  PLS_INTEGER  := 0;
0637:         v_cant_errores     PLS_INTEGER  := 0;
0638:         v_total_neto_aprob NUMBER(16,2) := 0;
0639:         v_total_reclamado  NUMBER(16,2) := 0;
0640:         v_neto_sin         NUMBER(14,2);
0641:     BEGIN
0642:         pr_log('INFO', NULL, 'Inicio lote de liquidacion. Fecha proceso: ' ||
0643:                              TO_CHAR(p_fecha_proceso, 'YYYY-MM-DD'));
0644: 
0645:         FOR r IN cur_pendientes LOOP
0646:             v_cant_procesados := v_cant_procesados + 1;
0647:             v_total_reclamado := v_total_reclamado + r.monto_reclamado;
0648: 
0649:             SAVEPOINT sp_siniestro;
0650:             BEGIN
0651:                 pr_liquidar_siniestro(r.siniestro_id, p_fecha_proceso);
0652: 
0653:                 SELECT estado INTO v_estado_final
0654:                   FROM ls_siniestros
0655:                  WHERE siniestro_id = r.siniestro_id;
0656: 
0657:                 IF v_estado_final = c_sin_liquidado THEN
0658:                     v_cant_liquidados := v_cant_liquidados + 1;
0659: 
0660:                     SELECT monto_neto INTO v_neto_sin
0661:                       FROM ls_liquidaciones
0662:                      WHERE siniestro_id = r.siniestro_id
0663:                        AND estado       = c_liq_aprobada;
0664: 
0665:                     v_total_neto_aprob := v_total_neto_aprob + v_neto_sin;
0666:                 ELSIF v_estado_final = c_sin_rechazado THEN
0667:                     v_cant_rechazados := v_cant_rechazados + 1;
0668:                 ELSIF v_estado_final = c_sin_pend_revision THEN
0669:                     v_cant_pendientes := v_cant_pendientes + 1;
0670:                 END IF;
0671:             EXCEPTION
0672:                 WHEN OTHERS THEN
0673:                     ROLLBACK TO sp_siniestro;
0674:                     v_cant_errores := v_cant_errores + 1;
0675:                     pr_log('ERROR', r.siniestro_id, SQLERRM);
0676:             END;
0677:         END LOOP;
0678: 
0679:         COMMIT;
0680: 
0681:         pr_log('INFO', NULL, 'Fin lote. Procesados=' || v_cant_procesados ||
0682:                              ' Liquidados=' || v_cant_liquidados ||
0683:                              ' Rechazados=' || v_cant_rechazados ||
0684:                              ' Pendientes=' || v_cant_pendientes ||
0685:                              ' Errores=' || v_cant_errores);
0686: 
0687:         DBMS_OUTPUT.PUT_LINE('=========================================');
0688:         DBMS_OUTPUT.PUT_LINE(' RESUMEN LOTE DE LIQUIDACION');
0689:         DBMS_OUTPUT.PUT_LINE(' Fecha de proceso   : ' || TO_CHAR(p_fecha_proceso, 'YYYY-MM-DD'));
0690:         DBMS_OUTPUT.PUT_LINE(' Procesados         : ' || v_cant_procesados);
0691:         DBMS_OUTPUT.PUT_LINE(' Liquidados         : ' || v_cant_liquidados);
0692:         DBMS_OUTPUT.PUT_LINE(' Rechazados         : ' || v_cant_rechazados);
0693:         DBMS_OUTPUT.PUT_LINE(' Pendientes revision: ' || v_cant_pendientes);
0694:         DBMS_OUTPUT.PUT_LINE(' Errores            : ' || v_cant_errores);
0695:         DBMS_OUTPUT.PUT_LINE(' Total neto aprobado: ' || TO_CHAR(v_total_neto_aprob, 'FM999999990.00'));
0696:         DBMS_OUTPUT.PUT_LINE('=========================================');
0697:     END pr_procesar_lote;
